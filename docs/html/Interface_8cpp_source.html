39void Interface::carregarProdutos()
40{
41 allProducts.clear();
42 std::ifstream file("data/produtos.txt");
43 std::string line;
44
45 while (std::getline(file, line))
46 {
47 if (line.empty())
48 continue;
49
50 std::stringstream ss(line);
51 std::string item;
52 std::vector<std::string> tokens;
53
54 while (std::getline(ss, item, '|'))
55 {
56 tokens.push_back(item);
57 }
58
59 if (tokens.size() >= 8)
60 {
61 try
62 {
63 unsigned int id = std::stoi(tokens[0]);
64 std::string nome = tokens[1];
65 std::string marca = tokens[2];
66 double valor = std::stod(tokens[3]);
67 double avaliacao = std::stod(tokens[4]);
68 unsigned int quantidade = std::stoi(tokens[5]);
69 std::string descricao = tokens[6];
70 std::string endereco = tokens[7];
71
72 allProducts.emplace_back(id, nome, marca, valor, avaliacao, quantidade, descricao, endereco);
73 }
74 catch (const std::exception &e)
75 {
76 // Skip malformed lines
77 continue;
78 }
79 }
80 }
81
82 displayedProducts = allProducts;
83}

288 wmove(product_win, 2, 12 + searchTerm.length());
289
290 c = wgetch(product_win);
291
292 if (searching)
293 {
294 if (c == 10) // Enter key - finish searching
295 {
296 searching = false;
297 if (static_cast<size_t>(highlight) >= displayedProducts.size() && !displayedProducts.empty())
298 {
299 highlight = displayedProducts.size() - 1;
300 }
301 else if (displayedProducts.empty())
302 {
303 highlight = 0;
304 }
305 }
306 else if (c == 27) // Escape key - clear search
307 {
308 searchTerm = "";
309 displayedProducts = filtrarProdutos(allProducts, searchTerm);
310 highlight = 0;
311 searching = false;
312 }
313 else if (c == KEY_BACKSPACE || c == 127 || c == 8)

316 {
317 searchTerm.pop_back();
318 displayedProducts = filtrarProdutos(allProducts, searchTerm);

323 {
324 searchTerm += c;
325 displayedProducts = filtrarProdutos(allProducts, searchTerm);
326 highlight = 0;
327 }
328 }
329 else
330 {
331 switch (c)
332 {
333 case KEY_UP:
334 if (!displayedProducts.empty())
335 {
336 if (highlight == 0)
337 {
338 highlight = displayedProducts.size() - 1;
339 }
340 else
341 {
342 highlight--;
343 }
344 }
345 break;
346 case KEY_DOWN:
347 if (!displayedProducts.empty())
348 {
349 if (static_cast<size_t>(highlight) == displayedProducts.size() - 1)
350 {
351 highlight = 0;
352 }
353 else
354 {
355 highlight++;
356 }
357 }
358 break;
359 case 10: // Enter key
360 if (!displayedProducts.empty() && static_cast<size_t>(highlight) < displayedProducts.size())
361 {
362 mostrarDetalhesView(displayedProducts[highlight]);

365 }
366 break;
367 case '/':
368 searching = true;
369 break;
370 case 'a':
371 case 'A':
372 if (!displayedProducts.empty() && static_cast<size_t>(highlight) < displayedProducts.size())
373 {
374 // Add to cart
375 usuarioLogado.carrinho.inserir(displayedProducts[highlight], 1);
376 mostrarMensagem("Produto adicionado ao carrinho!");
377 }
378 break;
379 case 'c':
380 case 'C':
381 mostrarJanelaCarrinho();
382 break;
383 case 'p':
384 case 'P':
385 mostrarJanelaPedidos();
386 break;
387 case 'e':
388 case 'E':
389 mostrarJanelaEditarPerfil();
390 break;
391 case 'n':
392 case 'N':
393 mostrarJanelaNovoProduto();
394 carregarProdutos(); // Reload products after adding new one
395 displayedProducts = filtrarProdutos(allProducts, searchTerm);
396 break;
397 case 'q':
398 case 'Q':
399 delwin(product_win);
400 delwin(profile_win);

433 mvwprintw(login_win, 3, 2, "Email: ");

435 wmove(login_win, 3, 9); // Start at email field

464 }

483 }
484 else if (isprint(ch) && user_len < 50)
485 {
486 username_c[user_len++] = ch;
487 waddch(login_win, ch);
488 }
489 break;
490 case State::PASSWORD:
491 if (ch == '\n')
492 {
493 state = State::DONE;
494 }
495 else if (ch == KEY_BACKSPACE || ch == 127 || ch == 8)
496 {
497 if (pass_len > 0)
498 {
499 pass_len--;
500 password_c[pass_len] = '\0';
501 mvwaddch(login_win, 4, 9 + pass_len, ' ');

506 {
507 password_c[pass_len++] = ch;
508 waddch(login_win, '*');
509 }
510 break;
511 case State::DONE:
512 break;
513 }
514 wrefresh(login_win);
515 }
516
517 // Try to login with the credentials
518 std::string email(username_c);
519 std::string senha(password_c);
520
521 bool loginSuccess = usuarioLogado.login(email, senha);
522
523 if (loginSuccess)

525 delwin(login_win);

527 }

529 {

531 mvwprintw(login_win, 7, 2, "Pressione qualquer tecla...");
