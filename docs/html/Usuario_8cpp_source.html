73std::vector<Produto> Usuario::obterProdutos()
74{
75 std::vector<Produto> produtos;
76 std::ifstream file("data/produtos.txt");
77 std::string line;
78
79 while (std::getline(file, line))
80 {
81 if (line.empty())
82 continue;
83
84 std::stringstream ss(line);
85 std::string item;
86 std::vector<std::string> tokens;
87
88 while (std::getline(ss, item, '|'))
89 {
90 tokens.push_back(item);
91 }
92
93 if (tokens.size() >= 8)
94 {
95 unsigned int id = std::stoi(tokens[0]);
96 std::string nome = tokens[1];
97 std::string marca = tokens[2];
98 double valor = std::stod(tokens[3]);
99 double avaliacao = std::stod(tokens[4]);
100 unsigned int quantidade = std::stoi(tokens[5]);
101 std::string descricao = tokens[6];
102 std::string endereco = tokens[7];
103
104 produtos.emplace_back(id, nome, marca, valor, avaliacao, quantidade, descricao, endereco);
105 }
106 }
107 file.close();
108 return produtos;
109}
